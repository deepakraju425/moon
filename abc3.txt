import pandas as pd
from google.colab import files

uploaded = files.upload()

df = pd.read_csv(next(iter(uploaded)))

# 1. Simple Random Sampling
srs = df.sample(n=10, random_state=42)
print("Simple Random Sampling:\n", srs)

# 2. Stratified Sampling
stratified = df.groupby('Extracurricular Activities').sample(
    frac=0.2, random_state=42
)
print("\nStratified Sampling:\n", stratified)

# 3. Cluster Sampling
df['Cluster'] = df.index // 1000
clusters = df['Cluster'].unique()
selected_cluster = pd.Series(clusters).sample(n=1, random_state=42).iloc[0]
cluster = df[df['Cluster'] == selected_cluster]
print("\nCluster Sampling:\n", cluster)
