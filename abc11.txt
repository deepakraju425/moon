#direct values
import numpy as np
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans

# Directly define feature values for 8 students
#col1 = hours studied, col2 = prev score, col3 = sleep hours
X = np.array([
    [2, 50, 4],
    [3, 55, 5],
    [4, 60, 6],
    [5, 65, 5],
    [7, 80, 8],
    [8, 85, 9],
    [9, 90, 7],
    [10, 95, 10]
])

# Standardize features because they have different scales
X_scaled = StandardScaler().fit_transform(X)

# Reduce 3 features to 2 principal components
pca = PCA(n_components=2)
X_pca = pca.fit_transform(X_scaled)

# Group the data into 2 clusters
kmeans = KMeans(n_clusters=2, random_state=42, n_init=10)
clusters = kmeans.fit_predict(X_pca)

# Visualize the clusters
plt.scatter(X_pca[:, 0], X_pca[:, 1], c=clusters)
plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")
plt.title("PCA and K-Means Clustering")
plt.show()

print("Principal Components:\n", X_pca)
print("Cluster Labels:\n", clusters)
print("Explained Variance Ratio:", pca.explained_variance_ratio_)
