
import numpy as np
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler

# Direct feature values for 8 students
#col1 = hours studied, col2 = prev score, col3 = sleep hours
X = np.array([
    [2, 50, 4],
    [3, 55, 5],
    [4, 60, 6],
    [5, 65, 5],
    [7, 80, 8],
    [8, 85, 9],
    [9, 90, 7],
    [10, 95, 10]
])

# Standardize features because they have different scales
X_scaled = StandardScaler().fit_transform(X)

# Create 2 clusters
# n_clusters=2 specifies the number of groups
# random_state=42 makes the results reproducible
kmeans = KMeans(n_clusters=2, random_state=42, n_init=10)

# Assign each student to a cluster
clusters = kmeans.fit_predict(X_scaled)

# Display cluster assignments
print("Cluster Labels:", clusters)

# Visualize clusters using the first two features
plt.scatter(X_scaled[:, 0], X_scaled[:, 1], c=clusters)
plt.xlabel("Hours Studied (Scaled)")
plt.ylabel("Previous Scores (Scaled)")
plt.title("K-Means Clustering")
plt.show()

# Display the cluster centers
print("Cluster Centers:\n", kmeans.cluster_centers_)
